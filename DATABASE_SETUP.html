Using the Database in Next.js

Install pg library

npm install pg

Create Database Utilities

Create src/lib/db.ts:

import { Client } from 'pg';

const client = new Client({
  connectionString: process.env.DATABASE_URL,
  ssl: {
    rejectUnauthorized: false,
  },
});

export async function connectDB() {
  if (!client.connected) {
    await client.connect();
  }
  return client;
}

export async function getPortfolioMetadata() {
  const db = await connectDB();
  const result = await db.query('SELECT * FROM portfolio_metadata LIMIT 1');
  return result.rows[0];
}

export async function getCaseStudies() {
  const db = await connectDB();
  const result = await db.query('SELECT * FROM case_studies WHERE status = $1 ORDER BY created_at DESC', ['published']);
  return result.rows;
}

export async function getHotspots(caseStudyId) {
  const db = await connectDB();
  const result = await db.query('SELECT * FROM hotspots WHERE case_study_id = $1', [caseStudyId]);
  return result.rows;
}

export async function getSkills() {
  const db = await connectDB();
  const result = await db.query('SELECT * FROM skills ORDER BY proficiency_level DESC');
  return result.rows;
}

Troubleshooting

Connection Error

Error: connect ECONNREFUSED

Solution: Check .env.local has correct DATABASE_URL

SSL Error

Error: no pg_hba.conf entry

Solution: Add ?sslmode=require to connection string (already included)

Timeout

Error: Client was closed by the server

Solution: - Check internet connection - Verify Neon credentials - Ensure sslmode is set to ‘require’
